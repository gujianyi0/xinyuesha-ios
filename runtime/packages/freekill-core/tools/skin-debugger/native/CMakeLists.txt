cmake_minimum_required(VERSION 3.22)
project(SkinInspector LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
file(TO_CMAKE_PATH "${FREEKILL_SOURCE_DIR}" FREEKILL_SOURCE_DIR)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick Concurrent)
if(NOT EXISTS "${FREEKILL_SOURCE_DIR}/src/ui/spine/spinebackend.h")
  message(FATAL_ERROR "Set FREEKILL_SOURCE_DIR to the FreeKill client source directory")
endif()

# Build an isolated copy of the game's adapters. The source tree is read-only.
# Only add metadata access; geometry and animation evaluation stay identical.
set(adapter "${CMAKE_CURRENT_BINARY_DIR}/adapters")
file(MAKE_DIRECTORY "${adapter}")
file(GLOB adapter_files "${FREEKILL_SOURCE_DIR}/src/ui/spine/spinebackend*"
                        "${FREEKILL_SOURCE_DIR}/src/ui/spine/spineversion.*"
                        "${FREEKILL_SOURCE_DIR}/src/ui/spine/texture.*")
foreach(source IN LISTS adapter_files)
  get_filename_component(name "${source}" NAME)
  configure_file("${source}" "${adapter}/${name}" COPYONLY)
endforeach()
file(READ "${adapter}/spinebackend.h" header)
string(REPLACE "virtual QRectF bounds() = 0;" "virtual QRectF bounds() = 0;\n    virtual QVariantList animationInfo() const = 0;" header "${header}")
file(WRITE "${adapter}/spinebackend.h" "#include <QVariantList>\n${header}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/animationinfo.inc" metadata)
foreach(name spinebackend_modern.inc spinebackend_21.cpp)
  file(READ "${adapter}/${name}" body)
  string(FIND "${body}" "public:" marker)
  if(marker LESS 0)
    message(FATAL_ERROR "Unsupported backend layout in ${name}")
  endif()
  string(REPLACE "public:" "public:\n${metadata}" body "${body}")
  file(WRITE "${adapter}/${name}" "${body}")
endforeach()

set(backends "${adapter}/spinebackend_factory.cpp")
foreach(version 21 34 35 36 37 38 40 41 42)
  set(runtime "${FREEKILL_SOURCE_DIR}/src/spine-versions/spine-c-${version}")
  file(GLOB runtime_sources CONFIGURE_DEPENDS "${runtime}/src/spine/*.c")
  if(NOT runtime_sources)
    message(FATAL_ERROR "Missing Spine runtime ${version}")
  endif()
  add_library(inspector_spine${version} STATIC ${runtime_sources})
  set_target_properties(inspector_spine${version} PROPERTIES AUTOMOC OFF POSITION_INDEPENDENT_CODE ON)
  target_include_directories(inspector_spine${version} PRIVATE "${runtime}/include")
  if(NOT MSVC)
    target_compile_options(inspector_spine${version} PRIVATE -w)
  endif()
  set_source_files_properties("${adapter}/spinebackend_${version}.cpp" PROPERTIES
    INCLUDE_DIRECTORIES "${runtime}/include")
  list(APPEND backends "${adapter}/spinebackend_${version}.cpp")
  list(APPEND runtime_libraries inspector_spine${version})
endforeach()
add_library(skininspector MODULE skeletonprobe.cpp skeletonprobe.h
  ${backends} "${adapter}/spineversion.cpp" "${adapter}/texture.cpp")
target_compile_definitions(skininspector PRIVATE QT_NO_KEYWORDS)
target_include_directories(skininspector PRIVATE "${adapter}")
target_link_libraries(skininspector PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Concurrent ${runtime_libraries})
set_target_properties(skininspector PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/module"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/module")
configure_file(qmldir "${CMAKE_CURRENT_BINARY_DIR}/module/qmldir" COPYONLY)
