cmake_minimum_required(VERSION 3.22)
project(IosFriendTransportTests LANGUAGES CXX OBJCXX)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_library(FOUNDATION_FRAMEWORK Foundation REQUIRED)
add_executable(test-ios-friends ../test_ios_friend_transport.mm ../../src/platform/iosfriendtransport.mm)
target_include_directories(test-ios-friends PRIVATE ../../src)
target_compile_features(test-ios-friends PRIVATE cxx_std_17)
set_target_properties(test-ios-friends PROPERTIES OBJCXX_STANDARD 17)
target_compile_options(test-ios-friends PRIVATE -fobjc-arc)
target_link_libraries(test-ios-friends PRIVATE Qt6::Core ${FOUNDATION_FRAMEWORK})
